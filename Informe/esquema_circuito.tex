% Wiring diagram of the physical system (Arduino UNO, servo, potentiometer, LEDs).
% Included from the main document with \input{esquema_circuito}.
\begin{circuitikz}[scale=0.82, transform shape, font=\small]
  % --- Arduino UNO block ---------------------------------------------------
  \draw[thick, fill=teal!8] (3,0) rectangle (6,6);
  \node[font=\bfseries] at (4.5,5.45) {Arduino UNO};
  \node[font=\footnotesize] at (4.5,5.0) {USB a PC};
  \node[anchor=west] at (3.05,4.5) {D5};
  \node[anchor=west] at (3.05,2.5) {D7};
  \node[anchor=east] at (5.95,4.5) {D9};
  \node[anchor=east] at (5.95,0.9) {A0};
  \node[anchor=south] at (3.8,0.05) {5V};
  \node[anchor=south] at (5.2,0.05) {GND};

  % Arduino power pins
  \draw (3.8,0) -- (3.8,-0.5) node[below] {$+5$\,V};
  \draw (5.2,0) -- (5.2,-0.4) node[ground] {};

  % --- Green LED on D5: ON while the gate has any opening -----------------
  \draw (3,4.5) to[R, l_=$220\,\Omega$, o-] (1,4.5)
        to[leDo, l_={LED verde}, color=green!50!black] (1,2.9)
        -- (1,2.6) node[ground] {};

  % --- Red LED on D7: ON only when the gate is fully closed ---------------
  \draw (3,2.5) to[short, o-] (2.3,2.5)
        to[R, l_=$220\,\Omega$] (2.3,0.9)
        to[leDo, l_={LED rojo}, color=red!70!black] (2.3,-1.0)
        -- (2.3,-1.3) node[ground] {};

  % --- Servo S3003 ----------------------------------------------------------
  \draw[thick, fill=violet!8] (8.2,3.8) rectangle (10.8,5.2);
  \node[font=\bfseries, align=center] at (9.5,4.5) {Servo\\S3003};
  \draw (6,4.5) to[short, o-] (8.2,4.5);
  \node[font=\footnotesize, anchor=south] at (7.1,4.55) {señal (naranja)};
  \draw (10.8,4.9) -- (11.4,4.9) node[right] {$+5$\,V (rojo)};
  \draw (10.8,4.1) -- (11.4,4.1) node[right] {GND (café)};

  % --- Potentiometer 10 k: wiper to A0, ends to +5 V and GND --------------
  \draw (8.6,-1.3) node[ground] {} to[pR, n=pot, l_={10\,k$\Omega$}] (8.6,1.9)
        -- (8.6,2.2) node[above] {$+5$\,V};
  \draw (6,0.9) to[short, o-] (7.2,0.9) |- (pot.wiper);
  \node[font=\footnotesize, anchor=south] at (6.6,0.95) {cursor};
\end{circuitikz}
